    10:   if thisObj == nil then
    11:     return false, nil
    12:   end
    13: 
    14:   local posOK, pos = pcall(function()
    15:     return thisObj:GetWorldPosition()
    16:   end)
    17: 
    18:   if not posOK or pos == nil then
    19:     return false, nil
    20:   end
    21: 
    22:   local dx = pos.x - (-64.850898742676)
    23:   local dy = pos.y - 12.987384796143
    24:   local dz = pos.z - 787.30694580078
    25:   local distanceSq = dx * dx + dy * dy + dz * dz
    26: 
    27:   return distanceSq <= 0.25, pos
    28: end

   168:     if ticket ~= nil and ticket ~= generation then return end

   183:       if attempt < 20 then

   185:           notify(source, attempt + 1, ticket)
